# Public terms ruleset. A hit anywhere in the tracked tree fails `npm run validate`.
# The patterns are regular expressions on purpose, so this file does not contain the literal text it blocks.
# An optional private list adds more terms: private/config/internal-terms.txt, or the file named by KIT_EXTRA_TERMS_FILE.
# Format: `re:` is a case-sensitive regex, `rei:` ignores case, anything else is a literal that ignores case and accents.

# Personal paths
re:/Us[e]rs/
re:/ho[m]e/[a-z][a-z0-9._-]*/
rei:[a-z]:\\Us[e]rs\\

# Email addresses (GitHub noreply and example domains are allowed)
rei:[a-z0-9._%+-]+@(?!noreply\.github\.com|users\.noreply\.github\.com|example\.|github\.com)[a-z0-9-]+(\.[a-z0-9-]+)*\.[a-z]{2,}

# Personal knowledge base names (tool names such as HashiCorp Vault stay allowed)
rei:obsidian[ _-]?vault
rei:obsidian://
rei:segundo[ _-]c[eé]rebro
rei:second[ _-]brain
rei:(?<!hashicorp )(?<!hashicorp-)(?<!key )(?<!ansible )(?<!azure )(?<!terraform, )(?<!products/)(?<!secret )\bvault\b
re:[^\s]\.agent[s]/

# Tokens and secrets
re:ghp_[A-Za-z0-9]{20,}
re:github_pat_[A-Za-z0-9_]{20,}
re:gho_[A-Za-z0-9]{20,}
re:sk-[A-Za-z0-9_-]{20,}
re:AKIA[0-9A-Z]{16}
re:xox[abprs]-[A-Za-z0-9-]{10,}
re:AIza[0-9A-Za-z_-]{30,}
rei:bearer\s+[a-z0-9._~+/-]{20,}
re:-----BEGIN [A-Z ]*PRIVATE KEY-----
